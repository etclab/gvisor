--- 6. the host side, while a sandbox lives ---------------------------------

$ $SCRATCH/bin/runsc $BASE run --bundle $SCRATCH/e1/bundle-sleep s1-sleep &
$ $SCRATCH/bin/runsc $BASE list
# comm is "exe" for every re-exec'd runsc process; argv[0] names them, so the
# processes were found by reading /proc/*/exe and /proc/*/cmdline:
$ for p in /proc/[0-9]*; do readlink $p/exe | grep -q bin/runsc && \
      echo "${p#/proc/} $(tr '\0' '\n' < $p/cmdline | head -1)"; done
$ cat /proc/<sandbox>/mounts /proc/<sandbox>/mountinfo /proc/<sandbox>/uid_map
$ grep -E '^(Uid|Gid|Cap|NoNewPrivs|Seccomp)' /proc/<sandbox>/status
$ capsh --decode=<CapEff>
$ ls -l /proc/<sandbox>/ns
# and the same for the gofer
$ wc -l /proc/mounts ; grep -c runsc /proc/mounts
  -> output-05-host-side.txt

--- 7. the mount table inside the sandbox -----------------------------------

$ $SCRATCH/bin/runsc $BASE run --bundle $SCRATCH/e1/bundle-mounts s1-inside \
      > $SCRATCH/e1/inside-mounts.txt 2>&1
  -> output-06-in-sandbox-mounts.txt
# It goes to its own file on purpose: runsc hands the caller's fd 1 straight to
# the sandbox, and writing both streams into one file clobbers the workload's
# output at the shared offset.
